CREATE TABLE Habitaciones (
	Nro_Hab INT AUTO_INCREMENT PRIMARY KEY,
    Tipo VARCHAR(25) NOT NULL,
    PrecioDia INT NOT NULL
);

CREATE TABLE alojamientos (
	CodAloj INT PRIMARY KEY,
    FechaIngreso DATE NOT NULL,
    NroDias INT NOT NULL,
    cod_cli INT NOT NULL,
    FOREIGN KEY (cod_cli) REFERENCES clientes(cod_cli)
);

CREATE TABLE HabitacionesAlojamiento (
	CodAloj INT NOT NULL,
    NroHab  INT NOT NULL,
    PRIMARY KEY (CodAloj, NroHab),
    FOREIGN KEY (CodAloj) REFERENCES alojamientos (CodAloj),
    FOREIGN KEY (NroHab) REFERENCES habitaciones(Nro_Hab)
);

-- Generar las siguiente consulta: Mostrar los datos de los alojamientos de meses impares, debe mostrar los siguientes datos:
-- NombreCliente  FechaIngreso  TipoHabitacion  NroDias  PrecioXDia  y el Cobro

SELECT
	c.nombre AS NombreCliente,
    a.FechaIngreso,
    h.Tipo AS TipoHabitacion,
    a.NroDias,
    h.PrecioDia AS PrecioXdia,
    (a.NroDias * h.PrecioDia) AS Cobro
FROM alojamientos a
JOIN clientes c ON a.cod_cli = c.cod_cli
JOIN HabitacionesAlojamiento ha ON a.CodAloj = ha.CodAloj
JOIN Habitaciones h ON ha.NroHab = h.Nro_Hab
WHERE MONTH(a.FechaIngreso) IN (1, 3, 5, 7, 9, 11)
ORDER BY a.FechaIngreso;
